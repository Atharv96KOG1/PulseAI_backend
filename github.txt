------------------------------------------------------------
1. FIRST-TIME SETUP
------------------------------------------------------------
# Clone the repo
git clone https://github.com/Atharv96KOG1/PulseAI_backend.git
cd PulseAI_backend

# Set your identity (once per machine)
git config --global user.name "Your Name"
git config --global user.email "you@example.com"

# Check current remotes
git remote -v


------------------------------------------------------------
2. STATUS / DIFF / LOG — look before you act
------------------------------------------------------------
git status                     # what's changed, staged vs unstaged
git diff                        # unstaged changes, file by file
git diff --staged               # staged changes (what a commit would include)
git log --oneline --graph -10   # last 10 commits, compact graph
git log -p backend/db.py        # full history of one file


------------------------------------------------------------
3. STAGING & COMMITTING
------------------------------------------------------------
git add backend/api/routes.py           # stage one file
git add backend/                        # stage a whole folder
git add -p                              # stage interactively, hunk by hunk
git status                              # confirm what's staged before committing

git commit -m "Add /report endpoint for PDF export"
git commit -am "Fix ruff line-length in pdf_report.py"   # add tracked changes + commit in one step
                                                          # (skips new/untracked files)

# Amend the LAST commit (only if not pushed yet)
git commit --amend -m "Add /report endpoint for weekly PDF export"


------------------------------------------------------------
4. BRANCHES
------------------------------------------------------------
git branch                              # list local branches
git branch -a                           # list local + remote branches
git branch -vv                          # local branches + what they track

git checkout -b feature/rag-pipeline    # create + switch to a new branch
git switch -c feature/rag-pipeline      # same thing, newer syntax

git switch main                         # switch to an existing branch
git checkout feature/schema-and-cli

git branch -d feature/old-experiment    # delete a LOCAL branch (safe, already merged)
git branch -D feature/old-experiment    # force-delete a local branch (unmerged, be sure first)

git push origin --delete feature/old-experiment   # delete a REMOTE branch


------------------------------------------------------------
5. PUSH
------------------------------------------------------------
git push origin feature/schema-and-cli          # push current branch to origin
git push -u origin feature/schema-and-cli       # first push of a new branch — sets upstream
git push                                        # after -u has been set once, just this works

# NEVER use these on shared branches unless you mean it and the team knows:
git push --force origin feature/schema-and-cli        # overwrites remote history — dangerous
git push --force-with-lease origin feature/schema-and-cli   # safer force: fails if someone
                                                             # else pushed since your last fetch


------------------------------------------------------------
6. PULL / FETCH
------------------------------------------------------------
git fetch origin                # download remote changes, don't merge yet
git fetch --all                 # fetch every remote

git pull                        # fetch + merge current branch's upstream
git pull origin main             # explicitly pull main into the branch you're on
git pull --rebase origin main    # replay your commits on top of main instead of merging
                                  # (keeps history linear — good before opening a PR)


------------------------------------------------------------
7. KEEPING A FEATURE BRANCH UP TO DATE WITH MAIN
------------------------------------------------------------
git switch main
git pull origin main
git switch feature/schema-and-cli
git rebase main                 # replay your commits on top of latest main
# ... resolve any conflicts, then:
git add <fixed files>
git rebase --continue
git push --force-with-lease origin feature/schema-and-cli   # rebase rewrites history,
                                                             # so a normal push is rejected

# Alternative if you prefer merge commits over rebase:
git switch feature/schema-and-cli
git merge main


------------------------------------------------------------
8. PULL REQUESTS (using GitHub CLI: `gh`)
------------------------------------------------------------
# Install once: https://cli.github.com   then:
gh auth login

# Create a PR from the current branch into main
gh pr create --base main --head feature/schema-and-cli \
  --title "Add RAG pipeline + PDF weekly report" \
  --body "Adds embedding-based Q&A over analyzed feedback and a PDF report export."

# Create a draft PR
gh pr create --draft --title "WIP: RAG grounding tweaks"

# List / view / check out PRs
gh pr list
gh pr view 12
gh pr checkout 12                # pull someone else's PR branch locally to test it

# Review & merge
gh pr review 12 --approve -b "LGTM, tested locally"
gh pr merge 12 --squash           # squash all commits into one on main
gh pr merge 12 --merge            # regular merge commit
gh pr merge 12 --rebase           # rebase-merge (linear history, no merge commit)

# (No `gh`? Same thing from the browser: push your branch, then GitHub shows
#  a "Compare & pull request" button on the repo page.)


------------------------------------------------------------
9. MERGING (without a PR, e.g. local integration branch)
------------------------------------------------------------
git switch main
git merge feature/schema-and-cli          # fast-forward or merge commit, whichever applies
git merge --no-ff feature/schema-and-cli  # always create a merge commit (keeps branch history visible)

# If a merge has conflicts:
#   1. Git marks conflicted files with <<<<<<< / ======= / >>>>>>> markers
#   2. Edit each file to resolve, then:
git add <resolved-file>
git commit                                 # completes the merge


------------------------------------------------------------
10. STASHING — shelve work without committing
------------------------------------------------------------
git stash                       # shelve tracked changes
git stash -u                    # also shelve untracked files
git stash list                  # see all stashes
git stash pop                   # re-apply the most recent stash and drop it
git stash apply stash@{1}       # re-apply a specific stash, keep it in the list
git stash drop stash@{1}        # delete a specific stash


------------------------------------------------------------
11. UNDOING THINGS (safest → most destructive)
------------------------------------------------------------
git restore backend/config.py             # discard unstaged changes to one file
git restore --staged backend/config.py    # unstage a file, keep the edits

git revert <commit-hash>                  # add a NEW commit that undoes an old one (safe on shared branches)

git reset --soft HEAD~1                   # undo last commit, keep changes staged
git reset --mixed HEAD~1                  # undo last commit, keep changes unstaged (default)
git reset --hard HEAD~1                   # undo last commit AND discard the changes — destructive,
                                           # only on commits nobody else has pulled

git clean -fd                             # delete untracked files/folders — destructive, check
                                           # `git status` and `git clean -nd` (dry run) first


------------------------------------------------------------
12. TAGS & RELEASES
------------------------------------------------------------
git tag v0.1.0                            # lightweight tag on current commit
git tag -a v0.1.0 -m "First working RAG + PDF report"   # annotated tag (preferred for releases)
git push origin v0.1.0                    # tags don't push automatically — push explicitly
git push origin --tags                    # push all tags at once

gh release create v0.1.0 --title "v0.1.0" --notes "Adds RAG Q&A and PDF export"


------------------------------------------------------------
13. INSPECTING WHAT CHANGED
------------------------------------------------------------
git diff main..feature/schema-and-cli         # everything this branch adds vs main
git log main..feature/schema-and-cli --oneline # just the commit list for that diff
git show <commit-hash>                         # full diff of one commit
git blame backend/rag/answer.py                # who last touched each line, and in which commit
